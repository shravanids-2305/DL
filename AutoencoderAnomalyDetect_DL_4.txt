------------------------------------------------------------------------------------
import tensorflow as tf
import numpy as np
import matplotlib.pyplot as plt

print("TensorFlow version:", tf.__version__)
------------------------------------------------------------------------------------
(x_train, y_train), (x_test, y_test) = tf.keras.datasets.mnist.load_data()
------------------------------------------------------------------------------------
print("Training data:", x_train.shape)
print("Testing data:", x_test.shape)
------------------------------------------------------------------------------------
x_train = x_train.astype("float32") / 255.0
x_test = x_test.astype("float32") / 255.0
------------------------------------------------------------------------------------
x_train = x_train.reshape((len(x_train), 784))
x_test = x_test.reshape((len(x_test), 784))

print("Training shape:", x_train.shape)
print("Testing shape:", x_test.shape)
------------------------------------------------------------------------------------
encoder = tf.keras.Sequential([
    tf.keras.layers.Input(shape=(784,)),
    tf.keras.layers.Dense(128, activation='relu'),
    tf.keras.layers.Dense(32, activation='relu'),
    tf.keras.layers.Dense(10, activation='relu')
])

encoder.summary()
------------------------------------------------------------------------------------
decoder = tf.keras.Sequential([
    tf.keras.layers.Input(shape=(10,)),
    tf.keras.layers.Dense(32, activation='relu'),
    tf.keras.layers.Dense(128, activation='relu'),
    tf.keras.layers.Dense(784, activation='sigmoid')
])

decoder.summary()
------------------------------------------------------------------------------------
autoencoder = tf.keras.Sequential([
    encoder,
    decoder
])

autoencoder.summary()
------------------------------------------------------------------------------------
autoencoder.compile(
    optimizer='adam',
    loss='mse',
    metrics=['mae']
)
------------------------------------------------------------------------------------
history = autoencoder.fit(
    x_train,
    x_train,
    epochs=10,
    batch_size=256,
    validation_data=(x_test, x_test)
)
------------------------------------------------------------------------------------
test_loss, test_mae = autoencoder.evaluate(x_test, x_test)

print("Test Loss:", test_loss)
print("Test MAE:", test_mae)
------------------------------------------------------------------------------------
plt.plot(history.history['loss'])
plt.plot(history.history['val_loss'])

plt.title("Autoencoder Loss")
plt.xlabel("Epoch")
plt.ylabel("Loss")

plt.legend(["Training Loss", "Validation Loss"])

plt.show()
------------------------------------------------------------------------------------
reconstructed = autoencoder.predict(x_test)
------------------------------------------------------------------------------------
plt.figure(figsize=(10, 4))

plt.subplot(1, 2, 1)
plt.imshow(x_test[0].reshape(28, 28), cmap='gray')
plt.title("Original Image")
plt.axis('off')

plt.subplot(1, 2, 2)
plt.imshow(reconstructed[0].reshape(28, 28), cmap='gray')
plt.title("Reconstructed Image")
plt.axis('off')

plt.show()
------------------------------------------------------------------------------------
reconstruction_error = np.mean(
    np.square(x_test - reconstructed),
    axis=1
)

print("First 10 reconstruction errors:")
print(reconstruction_error[:10])
------------------------------------------------------------------------------------
threshold = np.percentile(reconstruction_error, 95)

print("Anomaly threshold:", threshold)
------------------------------------------------------------------------------------
anomalies = reconstruction_error > threshold

print("Number of anomalies:", np.sum(anomalies))
print("Total test samples:", len(x_test))
------------------------------------------------------------------------------------
anomaly_index = np.argmax(reconstruction_error)

plt.imshow(x_test[anomaly_index].reshape(28, 28), cmap='gray')
plt.title(
    "Possible Anomaly\nError: "
    + str(round(reconstruction_error[anomaly_index], 4))
)
plt.axis('off')
plt.show()
