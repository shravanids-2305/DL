import tensorflow as tf
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import os

from PIL import Image
from tensorflow.keras.applications.vgg16 import VGG16, preprocess_input
from tensorflow.keras.layers import Input, GlobalAveragePooling2D
from tensorflow.keras.layers import Dense, Dropout
from tensorflow.keras.models import Model

print("TensorFlow Version:", tf.__version__)
print("All libraries imported successfully!")
------------------------------------------------------------------------------------
CSV_PATH = "VGG16_Object_Detection/dataset/annotations.csv"

df = pd.read_csv(CSV_PATH)

print("Annotations loaded successfully!")
display(df)
------------------------------------------------------------------------------------
IMAGE_DIR = "VGG16_Object_Detection/dataset/images"

image_size = 224

class_names = sorted(df["class"].unique())

class_mapping = {
    name: index for index, name in enumerate(class_names)
}

print("Class Mapping:", class_mapping)
------------------------------------------------------------------------------------
X = []
y_class = []
y_box = []

for index, row in df.iterrows():

    image_path = os.path.join(IMAGE_DIR, row["image"])

    img = Image.open(image_path).convert("RGB")

    original_width, original_height = img.size

    # Resize image
    img = img.resize((image_size, image_size))

    # Convert image to array
    img_array = np.array(img)

    X.append(img_array)

    # Convert class name to number
    y_class.append(class_mapping[row["class"]])

    # Normalize bounding box coordinates
    xmin = row["xmin"] / original_width
    ymin = row["ymin"] / original_height
    xmax = row["xmax"] / original_width
    ymax = row["ymax"] / original_height

    y_box.append([xmin, ymin, xmax, ymax])

X = np.array(X, dtype=np.float32)
y_class = np.array(y_class, dtype=np.int32)
y_box = np.array(y_box, dtype=np.float32)

print("Image array shape:", X.shape)
print("Class labels shape:", y_class.shape)
print("Bounding boxes shape:", y_box.shape)
------------------------------------------------------------------------------------
X = preprocess_input(X)

print("VGG16 preprocessing completed!")
------------------------------------------------------------------------------------
base_model = VGG16(
    weights="imagenet",
    include_top=False,
    input_shape=(224, 224, 3)
)

print("Pre-trained VGG16 model loaded successfully!")
------------------------------------------------------------------------------------
base_model.trainable = False

print("VGG16 layers frozen successfully!")

for layer in base_model.layers:
    print(layer.name, layer.trainable)
------------------------------------------------------------------------------------
input_layer = Input(shape=(224, 224, 3))

# VGG16 feature extraction
features = base_model(input_layer)

# Custom classifier
x = GlobalAveragePooling2D()(features)

x = Dense(256, activation="relu")(x)

x = Dropout(0.5)(x)

x = Dense(128, activation="relu")(x)

# Classification output
class_output = Dense(
    len(class_names),
    activation="softmax",
    name="class_output"
)(x)

# Bounding box output
box_output = Dense(
    4,
    activation="sigmoid",
    name="box_output"
)(x)

# Complete model
model = Model(
    inputs=input_layer,
    outputs=[class_output, box_output]
)

print("Custom object detection model created!")
------------------------------------------------------------------------------------
model.summary()
------------------------------------------------------------------------------------
model.compile(
    optimizer=tf.keras.optimizers.Adam(learning_rate=0.001),

    loss={
        "class_output": "sparse_categorical_crossentropy",
        "box_output": "mse"
    },

    metrics={
        "class_output": ["accuracy"]
    }
)

print("Model compiled successfully!")
------------------------------------------------------------------------------------
history = model.fit(
    X,
    {
        "class_output": y_class,
        "box_output": y_box
    },
    epochs=20,
    batch_size=2,
    verbose=1
)
------------------------------------------------------------------------------------
base_model.trainable = True

for layer in base_model.layers:

    if layer.name.startswith("block5"):
        layer.trainable = True

    else:
        layer.trainable = False

print("Last VGG16 convolutional block unfrozen!")

for layer in base_model.layers:
    print(layer.name, layer.trainable)
------------------------------------------------------------------------------------
model.compile(
